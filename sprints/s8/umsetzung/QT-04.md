
`qt04_p.gd` ist ein Wegwerf-Skript (nicht im Repo): lädt `res://start_menu.tscn`, sendet bei Frame 30 ein `InputEventKey` P (`Input.parse_input_event`, erreicht `start_menu._unhandled_input` Z. 136 bis 137), prüft bei Frame 60 und 300 `current_scene`, beendet bei Frame 360 mit Exit 0.

```gdscript
extends SceneTree
var frames := 0
func _initialize() -> void: change_scene_to_file("res://start_menu.tscn")
func _key(k: Key) -> void:
	var ev := InputEventKey.new(); ev.keycode = k; ev.physical_keycode = k; ev.pressed = true
	Input.parse_input_event(ev); Input.flush_buffered_events()
func _process(_dt: float) -> bool:
	frames += 1
	if frames == 30: _key(KEY_P)
	if frames == 60 or frames == 300: print("QT04 frame ", frames, ": scene=", current_scene.name)
	if frames == 360: print("QT04_OK"); quit(0)
	return false
```
